volantile

Der Operator volatile sagt dem Compiler, dass der Inhalt einer Variablen sich außerhalb des normalen Programmflusses ändern kann. Das kann zum Beispiel dann passieren, wenn ein Programm aus einer Interrupt-Service-Routine einen Wert erwartet und dann über diesen Wert einfach pollt (kein schönes Verhalten, aber gut zum Erklären von volatile). Siehe folgendes Beispiel

char keyPressed;
int count=0;

while (keyPressed != 'x') {
   count++;
}

Viele Compiler werden aus der while-Schleife ein while(1) machen,

volatile char keyPressed;
int count=0;

while (keyPressed != 'x') {
   count++;
}

Das Keyword volatile sollte sparsam verwendet werden, da es dem Compiler jegliches Optimieren verbietet.

Den Typ festlegen. Info für den Compiler
Nicht nur durch das Programm selbst, sondern auch durch andere Prozesse veränderbar sind.
Zb. Um nicht den Zugriff auf die Variable zu optimieren.

Extern const volantile unsigned clock_ticks;
—

- Funktionsregister, mit Statusfelder die sich ändern können, ohne das die CPU eingreift
Daher muss der Compiler wissen dass sie sich unabhängig vom Wissen des Compilers ändern können.
Der Complier weis nicht wer und wann diese bestimmte Variable liest/schreibt.
Der Compiler kann alle Optimierungen anwenden, die er möchte, solange er NICHT Annahmen darüber trifft, wann oder wie sich der Inhalt einer als “volantile” markierten Variable ändert. In der Praxis eliminiert dies viele Optimierungen.
——

volantile int foo;
Sagt dem Compiler, dass er explizit einen Lese-/Schreibvorgang für jede Anweisung, bei der es verwendet wird, erstellen soll. Das bedeutet, jedes Mal, wenn du foo verwendest, zieht der Compiler es von der Adresse, und wenn du etwas schreibst, schreibt er es sofort an die Adresse. Kein Caching in R1 und R2 oder im Stack! Es "schaltet keine Optimierung aus", die Optimierungen sind normalerweise mit diesem Schlüsselwort inkompatibel. Aber die Reihenfolge könnte sich trotzdem ändern.
Wann sollte man es verwenden? Du verwendest es, wenn sich das Objekt außerhalb des aktuellen Kontexts ändert. Dies kann ein anderer Thread sein, wie ein Interrupt, oder Hardwareregister. Der Standard bezeichnet es als "Änderung außerhalb der abstrakten Maschine", wenn ich mich recht
——
volatile bedeutet im Grunde genommen, dass 'Lesen und Schreiben von dieser Variablen Nebeneffekte sind', sodass ein volatile Lese- oder Schreibvorgang nicht mit einem anderen volatile Lese- oder Schreibvorgang (durch den Compiler: die Hardware kann tun, was sie will, und das ist der Hauptgrund, warum 'volatile' nicht als Multithreading-Primitiv verwendet werden sollte, außer auf einkern-Prozessoren, die keine Lese-/Schreibvorgänge neu anordnen).

Bei einem Fehler gibt diese Funktion ebenfalls NULL zurück. Ansonsten gibt sie eine Adresse der Struktur dirent zurück, die Folgendes beinhaltet:

struct dirent {
   long d_ino;      /* i-node Nr. (bei Windows/MS-DOS immer 0) */
   unsigned short d_reclen;   /* (bei Windows/MS-DOS immer 0)  */
   unsigned short d_namlen;   /* Länge des Namens in d_name    */
   char *d_name;       /* Dateiname mit abschließendem '\0'    */
};

In der Praxis kann die Funktion readdir() so verwendet werden:

/* read_dir.c */
#include <sys/types.h>
#include <dirent.h>
#include <stdio.h>
#include <stdlib.h>

int main(int argc, char *argv[]) {
   DIR *dir;
   struct dirent *dirzeiger;

   if(argc != 2) {
      fprintf(stderr,"Benutzung : %s Directory\n", argv[0]);
      return EXIT_FAILURE;
   }
   /* das Verzeichnis öffnen */
   if((dir=opendir(argv[1])) == NULL) {
      fprintf(stderr,"Fehler bei opendir ...\n");
      return EXIT_FAILURE;
   }
   /* das komplette Verzeichnis auslesen */
   while((dirzeiger=readdir(dir)) != NULL)
      printf("%s\n",(*dirzeiger).d_name);
   /* Lesezeiger wieder schließen */
   if(closedir(dir) == -1)
      printf("Fehler beim Schließen von %s\n", argv[1]);
   return EXIT_SUCCESS;
}

volantile

Der Operator volatile sagt dem Compiler, dass der Inhalt einer Variablen sich außerhalb des normalen Programmflusses ändern kann. Das kann zum Beispiel dann passieren, wenn ein Programm aus einer Interrupt-Service-Routine einen Wert erwartet und dann über diesen Wert einfach pollt (kein schönes Verhalten, aber gut zum Erklären von volatile). Siehe folgendes Beispiel

char keyPressed;
int count=0;

while (keyPressed != 'x') {
   count++;
}

Viele Compiler werden aus der while-Schleife ein while(1) machen,

volatile char keyPressed;
int count=0;

while (keyPressed != 'x') {
   count++;
}

Das Keyword volatile sollte sparsam verwendet werden, da es dem Compiler jegliches Optimieren verbietet.

Den Typ festlegen. Info für den Compiler
Nicht nur durch das Programm selbst, sondern auch durch andere Prozesse veränderbar sind.
Zb. Um nicht den Zugriff auf die Variable zu optimieren.

Extern const volantile unsigned clock_ticks;
—

- Funktionsregister, mit Statusfelder die sich ändern können, ohne das die CPU eingreift
Daher muss der Compiler wissen dass sie sich unabhängig vom Wissen des Compilers ändern können.
Der Complier weis nicht wer und wann diese bestimmte Variable liest/schreibt.
Der Compiler kann alle Optimierungen anwenden, die er möchte, solange er NICHT Annahmen darüber trifft, wann oder wie sich der Inhalt einer als “volantile” markierten Variable ändert. In der Praxis eliminiert dies viele Optimierungen.
——

volantile int foo;
Sagt dem Compiler, dass er explizit einen Lese-/Schreibvorgang für jede Anweisung, bei der es verwendet wird, erstellen soll. Das bedeutet, jedes Mal, wenn du foo verwendest, zieht der Compiler es von der Adresse, und wenn du etwas schreibst, schreibt er es sofort an die Adresse. Kein Caching in R1 und R2 oder im Stack! Es "schaltet keine Optimierung aus", die Optimierungen sind normalerweise mit diesem Schlüsselwort inkompatibel. Aber die Reihenfolge könnte sich trotzdem ändern.
Wann sollte man es verwenden? Du verwendest es, wenn sich das Objekt außerhalb des aktuellen Kontexts ändert. Dies kann ein anderer Thread sein, wie ein Interrupt, oder Hardwareregister. Der Standard bezeichnet es als "Änderung außerhalb der abstrakten Maschine", wenn ich mich recht
——
volatile bedeutet im Grunde genommen, dass 'Lesen und Schreiben von dieser Variablen Nebeneffekte sind', sodass ein volatile Lese- oder Schreibvorgang nicht mit einem anderen volatile Lese- oder Schreibvorgang (durch den Compiler: die Hardware kann tun, was sie will, und das ist der Hauptgrund, warum 'volatile' nicht als Multithreading-Primitiv verwendet werden sollte, außer auf einkern-Prozessoren, die keine Lese-/Schreibvorgänge neu anordnen).

int main(int argc, char *argv[]) {
   DIR *dir;
   struct dirent *dirzeiger;

   if(argc != 2) {
      fprintf(stderr,"Benutzung : %s Directory\n", argv[0]);
      return EXIT_FAILURE;
   }
   /* das Verzeichnis öffnen */
   if((dir=opendir(argv[1])) == NULL) {
      fprintf(stderr,"Fehler bei opendir ...\n");
      return EXIT_FAILURE;
   }
   /* das komplette Verzeichnis auslesen */
   while((dirzeiger=readdir(dir)) != NULL)
      printf("%s\n",(*dirzeiger).d_name);
   /* Lesezeiger wieder schließen */
   if(closedir(dir) == -1)
      printf("Fehler beim Schließen von %s\n", argv[1]);
   return EXIT_SUCCESS;
}

	// Output
	print_button(power);

	// Assign: Function switch ON/OFF
	power = press(power);

	// Output
	print_button(power);


	return 0;

// Function: switch ON/OFF
Button press(Button current)
{
	if (current == ON) return OFF;
	else return ON;
}

// Function: Output
void print_button(Button current)
{
	if (current == ON) printf(“ON\n”);
	else if (current == OFF) printf(“OFF\”);
}

#include <stdio.h>      /* for printf and stderr */
#include <string.h>     /* for strerror */
#include <dirent.h>     /* for DIR, opendir, readdir, and dirent */
#include <errno.h>      /* for errno */
#include <sys/stat.h>   /* for stat */

int main(int argc, char **argv)
{
    DIR*           FD;        /* represent the directory */
    struct dirent* in_file;   /* represent the file */
    char*          target_dir = "."; /* current directory */

    /* Scanning the target directory */
    FD = opendir(target_dir);
    if (FD == NULL) 
    {
        fprintf(stderr, "Error: Failed to open input directory - %s\n", strerror(errno));
        return 1;
    }

    /* Reading object (files, directories ...) from the folder */
    while ((in_file = readdir(FD)))
    {
        struct stat buffer;
        int         status;

		// in_file?
        status = stat(in_file->d_name, &buffer);

        /* check status */
        if (status == -1)
        {
            fprintf(stderr, "Error: Failed to stat item - %s\n", strerror(errno));
            return 1;
        }

        /* check result */
        if ( buffer.st_mode & S_IFREG )
        {
            printf("%s is file \n", in_file->d_name);
        }
    }
    
    /* Close the directory */
    closedir(FD);
    
    return 0;
}

#include <stdio.h>
#include <string.h>

int main(void) {
    const char *full_path = "/home/user/documents/report.txt";

    // Find the last slash to isolate the filename
    const char *fname = strrchr(full_path, '/'); 
    if (!fname) fname = full_path; // No slash found, the whole path is the filename
    else fname++;                  // Move past the slash

    // Find the last dot to isolate the extension
    const char *ext = strrchr(fname, '.');

    printf("Filename with ext: %s\n", fname); // report.txt
    if (ext) printf("Extension: %s\n", ext);  // .txt
